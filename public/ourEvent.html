document.addEventListener("DOMContentLoaded", () => { loadEvents(); }); async function loadEvents() { try { const response = await fetch("/api/events"); const result = await response.json(); if (!result.success) { console.error(result.message); return; } const eventContainer = document.getElementById("eventContainer"); eventContainer.innerHTML = ""; result.data.forEach(event => { const card = document.createElement("div"); card.className = "col-lg-4 col-md-6"; card.innerHTML = `

${event.title || ""}

${event.description || ""}

${event.event_date || ""} ${event.location || ""}
`; eventContainer.appendChild(card); }); } catch (error) { console.error("Events loading error:", error); } }
