document.addEventListener("DOMContentLoaded", () => { loadBlogs(); }); async function loadBlogs() { try { const response = await fetch("/api/blogs"); const result = await response.json(); if (!result.success) { console.error(result.message); return; } const blogContainer = document.getElementById("blogContainer"); blogContainer.innerHTML = ""; result.data.forEach(blog => { const card = document.createElement("div"); card.className = "col-lg-4 col-md-6"; card.innerHTML = `
${blog.category_name || ""}

${blog.title || ""}

${blog.description || ""}

${blog.blog_date || ""}
`; blogContainer.appendChild(card); }); } catch (error) { console.error("Blogs loading error:", error); } }
